  <script>
    // ----- Hearts background (Gate only)
    const hearts = document.getElementById('hearts');
    for (let i=0;i<18;i++){
      const h=document.createElement('div');
      h.className='heart';
      h.style.left = Math.random()*100 + '%';
      h.style.bottom = (-20 - Math.random()*120) + 'px';
      h.style.animationDelay = (Math.random()*5) + 's';
      h.style.animationDuration = (5.5 + Math.random()*4) + 's';
      h.style.opacity = (0.3 + Math.random()*0.6);
      h.style.transform = `scale(${0.7 + Math.random()*1.2}) rotate(45deg)`;
      hearts.appendChild(h);
    }

    // ----- Section navigation
    const sections = Array.from(document.querySelectorAll('.section'));
    function show(id){
      sections.forEach(s => s.classList.toggle('active', s.id===id));
      window.scrollTo({top:0, behavior:'smooth'});
      // Try to start music when entering love section
      if(id==='s-love'){
        attemptUnmuteOnFirstTouch();
      }
    }
    document.querySelectorAll('[data-next]').forEach(btn=>{
      btn.addEventListener('click', ()=> show(btn.dataset.next));
    });

    // ----- Password Gate
    const correct = '2025/2/8';
    const pw = document.getElementById('pw');
    const errBox = document.getElementById('errBox');
    function normalize(x){
      return (x||'').trim()
        .replace(/\s+/g,'')
        .replace(/-/g,'/')
        .replace(/\\/g,'/');
    }
    document.getElementById('unlockBtn').addEventListener('click', ()=>{
      const v = normalize(pw.value);
      if(v === correct){
        errBox.classList.add('hidden');
        show('s-momen');
      }else{
        errBox.classList.remove('hidden');
      }
    });
    pw.addEventListener('keydown', (e)=>{
      if(e.key==='Enter') document.getElementById('unlockBtn').click();
    });

    // ----- Timer since 2025/2/8 (local time)
    const start = new Date(2025, 1, 8, 0, 0, 0); // month is 0-based; 1 = February
    const elD = document.getElementById('d');
    const elH = document.getElementById('h');
    const elM = document.getElementById('m');
    const elS = document.getElementById('s');

    function tick(){
      const now = new Date();
      let diff = Math.max(0, now - start);
      const sec = Math.floor(diff/1000);
      const days = Math.floor(sec / 86400);
      const hours = Math.floor((sec % 86400) / 3600);
      const mins = Math.floor((sec % 3600) / 60);
      const secs = sec % 60;
      elD.textContent = days;
      elH.textContent = hours;
      elM.textContent = mins;
      elS.textContent = secs;
    }
    tick();
    setInterval(tick, 1000);

    // ----- YouTube sound handling
    const yt = document.getElementById('yt');
    const soundBtn = document.getElementById('soundBtn');

    function setYTUnmuted(){
      // Reload iframe with mute=0 (browser may require user gesture)
      const base = "https://www.youtube-nocookie.com/embed/5FlF1evtCFg";
      yt.src = base + "?autoplay=1&mute=0&playsinline=1&rel=0";
    }
    soundBtn.addEventListener('click', ()=>{
      setYTUnmuted();
    });

    // On first user interaction anywhere, try unmute once
    let unmuteTried = false;
    function attemptUnmuteOnFirstTouch(){
      if(unmuteTried) return;
      const handler = ()=>{
        if(unmuteTried) return;
        unmuteTried = true;
        setYTUnmuted();
        document.removeEventListener('click', handler);
        document.removeEventListener('touchstart', handler);
      };
      document.addEventListener('click', handler, {once:true});
      document.addEventListener('touchstart', handler, {once:true});
    }
  </script>
